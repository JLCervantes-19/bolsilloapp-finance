-- ---------------------------------------------------------------------------
-- profiles — 1:1 con auth.users
-- ---------------------------------------------------------------------------
create table public.profiles (
  id          uuid primary key references auth.users(id) on delete cascade,
  full_name   text,
  avatar_url  text,
  currency    text not null default 'COP',
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

comment on table public.profiles is 'Perfil de usuario, 1:1 con auth.users.';

-- ---------------------------------------------------------------------------
-- categories — categorías de ingreso/gasto, por defecto y personalizadas
-- ---------------------------------------------------------------------------
create table public.categories (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id) on delete cascade,
  name        text not null,
  color       text not null,
  kind        text not null check (kind in ('income', 'expense')),
  is_default  boolean not null default false,
  created_at  timestamptz not null default now(),
  updated_at  timestamptz not null default now()
);

comment on table public.categories is 'Categorías de ingreso/gasto por usuario, sembradas por defecto al registrarse.';

-- ---------------------------------------------------------------------------
-- incomes — ingresos fijos, variables, recurrentes
-- ---------------------------------------------------------------------------
create table public.incomes (
  id           uuid primary key default gen_random_uuid(),
  user_id      uuid not null references auth.users(id) on delete cascade,
  category_id  uuid references public.categories(id) on delete set null,
  amount       numeric(14,2) not null check (amount > 0),
  description  text,
  source       text,
  date         date not null default current_date,
  recurrence   text not null default 'variable' check (recurrence in ('fixed', 'variable', 'recurring')),
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now()
);

comment on table public.incomes is 'Ingresos: salario, roll-over, ingresos extra, rendimientos.';

-- ---------------------------------------------------------------------------
-- expenses — gastos fijos y variables
-- ---------------------------------------------------------------------------
create table public.expenses (
  id              uuid primary key default gen_random_uuid(),
  user_id         uuid not null references auth.users(id) on delete cascade,
  category_id     uuid references public.categories(id) on delete set null,
  amount          numeric(14,2) not null check (amount > 0),
  description     text,
  payment_method  text,
  tags            text[] not null default '{}',
  is_fixed        boolean not null default false,
  date            date not null default current_date,
  created_at      timestamptz not null default now(),
  updated_at      timestamptz not null default now()
);

comment on table public.expenses is 'Gastos fijos (arriendo, servicios, suscripciones) y variables (mercado, transporte, antojos).';

-- ---------------------------------------------------------------------------
-- debts — deudas fijas/variables, motor snowball/avalanche
-- ---------------------------------------------------------------------------
create table public.debts (
  id                uuid primary key default gen_random_uuid(),
  user_id           uuid not null references auth.users(id) on delete cascade,
  name              text not null,
  balance           numeric(14,2) not null check (balance >= 0),
  original_balance  numeric(14,2) not null,
  interest_rate     numeric(5,2) not null default 0,
  minimum_payment   numeric(14,2) not null default 0,
  term_months       integer,
  due_date          date,
  priority          integer not null default 0,
  status            text not null default 'active' check (status in ('active', 'paid_off')),
  created_at        timestamptz not null default now(),
  updated_at        timestamptz not null default now()
);

comment on table public.debts is 'Deudas del usuario — insumo del Debt Engine (bola de nieve / avalancha).';

-- ---------------------------------------------------------------------------
-- budgets — topes mensuales por categoría
-- ---------------------------------------------------------------------------
create table public.budgets (
  id           uuid primary key default gen_random_uuid(),
  user_id      uuid not null references auth.users(id) on delete cascade,
  category_id  uuid not null references public.categories(id) on delete cascade,
  month        date not null,
  cap_amount   numeric(14,2) not null check (cap_amount >= 0),
  created_at   timestamptz not null default now(),
  updated_at   timestamptz not null default now(),
  unique (user_id, category_id, month)
);

comment on table public.budgets is 'Tope mensual de gasto por categoría. month = primer día del mes (YYYY-MM-01).';

-- ---------------------------------------------------------------------------
-- alerts — avisos y notificaciones generadas por el Recommendation Engine
-- ---------------------------------------------------------------------------
create table public.alerts (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id) on delete cascade,
  type        text not null,
  tone        text not null default 'positive' check (tone in ('positive', 'warning', 'negative')),
  message     text not null,
  is_read     boolean not null default false,
  created_at  timestamptz not null default now()
);

comment on table public.alerts is 'Insight cards generadas por el Recommendation & Alert Engine (backend), consumidas por el dashboard.';

-- ---------------------------------------------------------------------------
-- goals — metas de ahorro, incluye fondo de emergencia
-- ---------------------------------------------------------------------------
create table public.goals (
  id                  uuid primary key default gen_random_uuid(),
  user_id             uuid not null references auth.users(id) on delete cascade,
  name                text not null,
  target_amount       numeric(14,2) not null check (target_amount > 0),
  current_amount      numeric(14,2) not null default 0,
  target_date         date,
  color               text,
  is_emergency_fund   boolean not null default false,
  created_at          timestamptz not null default now(),
  updated_at          timestamptz not null default now()
);

comment on table public.goals is 'Metas de ahorro. is_emergency_fund marca la meta que representa el fondo de emergencia.';

-- ---------------------------------------------------------------------------
-- savings — abonos de ahorro (emergencia, automáticos, manuales)
-- ---------------------------------------------------------------------------
create table public.savings (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id) on delete cascade,
  goal_id     uuid references public.goals(id) on delete set null,
  amount      numeric(14,2) not null check (amount > 0),
  type        text not null default 'manual' check (type in ('emergency', 'automatic', 'manual')),
  date        date not null default current_date,
  created_at  timestamptz not null default now()
);

comment on table public.savings is 'Movimientos de ahorro; puede o no estar ligado a una meta específica.';

-- ---------------------------------------------------------------------------
-- investments — portafolios, rendimientos, activos
-- ---------------------------------------------------------------------------
create table public.investments (
  id                uuid primary key default gen_random_uuid(),
  user_id           uuid not null references auth.users(id) on delete cascade,
  name              text not null,
  type              text,
  amount_invested   numeric(14,2) not null check (amount_invested >= 0),
  current_value     numeric(14,2) not null check (current_value >= 0),
  date              date not null default current_date,
  created_at        timestamptz not null default now(),
  updated_at        timestamptz not null default now()
);

comment on table public.investments is 'Inversiones del usuario — alimenta la card de Inversiones y Patrimonio.';

-- ---------------------------------------------------------------------------
-- settings — configuración de app por usuario
-- ---------------------------------------------------------------------------
create table public.settings (
  user_id                 uuid primary key references auth.users(id) on delete cascade,
  theme                   text not null default 'light' check (theme in ('light', 'dark')),
  currency                text not null default 'COP',
  locale                  text not null default 'es-CO',
  notifications_enabled   boolean not null default true,
  created_at              timestamptz not null default now(),
  updated_at              timestamptz not null default now()
);

comment on table public.settings is 'Configuración de la app por usuario (tema, moneda, idioma, notificaciones).';

-- ---------------------------------------------------------------------------
-- audit_logs — historial de seguridad y transacciones
-- ---------------------------------------------------------------------------
create table public.audit_logs (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id) on delete cascade,

  created_at  timestamptz not null default now()

comment on table public.audit_logs is 'Historial de auditoría, poblado por triggers en expenses/incomes/debts/goals.';

-- ---------------------------------------------------------------------------
-- 3.1 updated_at genérico
-- ---------------------------------------------------------------------------
create or replace function public.handle_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

create trigger set_updated_at before update on public.profiles
  for each row execute function public.handle_updated_at();
create trigger set_updated_at before update on public.categories
  for each row execute function public.handle_updated_at();
create trigger set_updated_at before update on public.incomes
  for each row execute function public.handle_updated_at();

  for each row execute function public.handle_updated_at();

-- ---------------------------------------------------------------------------
-- 3.2 audit log genérico — aplicado a tablas transaccionales
-- ---------------------------------------------------------------------------
create or replace function public.log_audit()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid;
  v_entity_id uuid;
begin
  v_user_id := coalesce(new.user_id, old.user_id);
  v_entity_id := coalesce(new.id, old.id);

  insert into public.audit_logs (user_id, action, entity, entity_id, metadata)
  values (
    v_user_id,
    lower(TG_OP),
    TG_TABLE_NAME,
    v_entity_id,
    case TG_OP
      when 'DELETE' then to_jsonb(old)
      else to_jsonb(new)
    end
  );

  return coalesce(new, old);
end;
$$;

create trigger audit_expenses after insert or update or delete on public.expenses
  for each row execute function public.log_audit();
create trigger audit_incomes after insert or update or delete on public.incomes
  for each row execute function public.log_audit();
create trigger audit_debts after insert or update or delete on public.debts
  for each row execute function public.log_audit();
create trigger audit_goals after insert or update or delete on public.goals
  for each row execute function public.log_audit();

-- ---------------------------------------------------------------------------
-- 3.3 alta de usuario — perfil, settings y categorías por defecto
-- ---------------------------------------------------------------------------
create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  insert into public.profiles (id, full_name, avatar_url)
  values (
    new.id,
    new.raw_user_meta_data ->> 'full_name',
    new.raw_user_meta_data ->> 'avatar_url'
  );

  insert into public.settings (user_id) values (new.id);

  insert into public.categories (user_id, name, color, kind, is_default) values
    (new.id, 'Vivienda',       '#2B2D42', 'expense', true),
    (new.id, 'Servicios',      '#E76F51', 'expense', true),
    (new.id, 'Alimentación',   '#F4A261', 'expense', true),
    (new.id, 'Suscripciones',  '#E07A5F', 'expense', true),
    (new.id, 'Transporte',     '#81B29A', 'expense', true),
    (new.id, 'Antojos',        '#787774', 'expense', true),
    (new.id, 'Salario',        '#2A9D8F', 'income',  true),
    (new.id, 'Ingreso extra',  '#81B29A', 'income',  true),
    (new.id, 'Rendimientos',   '#F4A261', 'income',  true);

  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();
